// API

Три строки до продакшена

SDK для любимого языка. Деплойте из CI, скриптов или прямо из кода.

  • Типизированный SDK для TypeScript и Python
  • REST API с идемпотентными запросами
  • Вебхуки на каждое событие деплоя
import { Forge } from '@forge/sdk'

const forge = new Forge({ token: process.env.FORGE_TOKEN })

const deploy = await forge.deploy({
  project: 'shop',
  target: 'production',
  regions: ['ala1', 'fra1', 'sin1'],
})
// → https://shop.forge.app (24.6s)
from forge import Forge

forge = Forge(token=os.environ["FORGE_TOKEN"])

deploy = forge.deploy(
    project="shop",
    target="production",
)
print(deploy.url)  # https://shop.forge.app
